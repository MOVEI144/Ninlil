cmake_minimum_required(VERSION 3.20)
project(ninlil_consumer LANGUAGES C)
option(CONSUMER_SECURE "Exercise the optional secure package" ON)
if(CONSUMER_SECURE)
  find_package(Ninlil 0.1 REQUIRED COMPONENTS core secure bulk)
else()
  find_package(Ninlil 0.1 REQUIRED COMPONENTS core)
endif()
# Multiple independent consumers may discover the same package in one build.
if(DEFINED CONSUMER_SWITCH_BACKEND)
  set(Ninlil_JOURNAL_BACKEND "${CONSUMER_SWITCH_BACKEND}")
endif()
find_package(Ninlil 0.1 REQUIRED)
add_executable(consumer main.c)
target_compile_features(consumer PRIVATE c_std_11)
target_compile_options(consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(consumer PRIVATE Ninlil::${Ninlil_JOURNAL_BACKEND})
if(CONSUMER_SECURE)
  target_compile_definitions(consumer PRIVATE CONSUMER_SECURE=1)
  target_link_libraries(consumer PRIVATE Ninlil::node Ninlil::identity_file Ninlil::bulk)
endif()
